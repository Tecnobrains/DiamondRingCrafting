        timedAction = Making,
        category = Jewelry,

        }

        outputs

        itemMapper RingMapper

            Base.Ring_Left_RingFinger_Gold = Base.Ring_Left_RingFinger_GoldDiamond,
            Base.Ring_Right_RingFinger_Gold = Base.Ring_Right_RingFinger_GoldDiamond,
            Base.Ring_Left_MiddleFinger_Gold = Base.Ring_Left_MiddleFinger_GoldDiamond,
            Base.Ring_Right_MiddleFinger_Gold = Base.Ring_Right_MiddleFinger_GoldDiamond,

            default = Base.Ring_Left_RingFinger_GoldDiamond,
        }
    }
}
